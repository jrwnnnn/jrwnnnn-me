SELECT 'article' AS type,
    title, 
    subtitle, 
    content AS body,
    NULL AS notes, 
    NULL AS rating, 
    art AS cover, 
    '/transmission/' || slug AS url,
    date_created
FROM articles

UNION ALL

SELECT 'rating' AS type,
    title, 
    NULL AS subtitle, 
    NULL AS body,
    notes, 
    rating, 
    art AS cover, 
    url, 
    date_created
FROM ratings

ORDER BY date_created DESC
